# Explicit version pins for Odoo 14.0 / Python 3.10 builds.
# Each line replaces the aggregated constraint for that package.
# Use when aggregated constraints produce impossible combinations
# (e.g. an old repo pins <X while a newer one requires >=X).

# cryptography: old OCA repos pin <23.2, new ones require >=43.
# Odoo 14 + paramiko + pyOpenSSL work fine with 42.x.
cryptography>=38.0.0,<43.0.0

# lxml: 5.2 (2024-04) moved lxml.html.clean into a separate lxml_html_clean
# package. Odoo 14's odoo/tools/mail.py does `from lxml.html import clean`
# and crashes on import with lxml>=5.2 ("lxml.html.clean module is now a
# separate project"). Odoo 17+ handle the split; 14/15/16 need the old lxml.
lxml>=4.9.0,<5.2.0

# Odoo 14 imports PyPDF2.utils and the legacy PdfFile* API. PyPDF2 3.x
# removed those compatibility symbols, so keep the last compatible major.
PyPDF2>=1.26.0,<2.0.0

# Odoo 14 imports the legacy URL helpers removed in Werkzeug 2.1.
Werkzeug>=1.0.1,<2.0.0
